// ════════════════════════════════════════════════════
//   CODE MAP — Apps Script for the restored dashboard version
//   SHARED CONFIG: sheet names, headers, JSON output
//   WEB APP ROUTER: doGet/doPost action dispatch only
//   ACTIVITIES API: Activities sheet -> Transactions + Dashboard flows
//   WHEEL API: Wheel_orders / Wheel_lots / Wheel_history -> Wheel page
//   SHEET HELPERS: table read/write, parsing, date formatting
//   CROSS-PAGE DATA:
//     Activities -> Transactions page and Dashboard totals
//     Wheel_* -> Wheel page only
//     Settings are local/default in index.html for this restored version
// ════════════════════════════════════════════════════

// ════════════════════════════════════════════════════
//   SHARED CONFIG
// ════════════════════════════════════════════════════

const SHEET_ACTIVITIES = 'Activities';
const SHEET_WHEEL_ORDERS = 'Wheel_orders';
const SHEET_WHEEL_LOTS = 'Wheel_lots';
const SHEET_WHEEL_HISTORY = 'Wheel_history';
const CODE_VERSION = '2026-06-03-restored-stable';

const ACTIVITY_HEADERS = ['date', 'stocks', 'bonds', 'crypto', 'gold', 'currency', 'note'];
const WHEEL_ORDER_HEADERS = ['id', 'cex', 'product', 'openDate', 'expiryDate', 'strike', 'contracts', 'premium', 'currency', 'status', 'note', 'exercisePL', 'updatedAt'];
const WHEEL_LOT_HEADERS = ['id', 'name', 'price', 'amount', 'target', 'accRealPL', 'problem', 'updatedAt'];

// ════════════════════════════════════════════════════
//   WEB APP ROUTER
// ════════════════════════════════════════════════════

function doGet(e) {
  const action = String((e && e.parameter && e.parameter.action) || '').toLowerCase();
  try {
    if (action === 'activities_write') return jsonOutput_(appendActivityFromParams_(e.parameter || {}));
    if (action === 'activities_read') return jsonOutput_(readActivities_());
    if (action === 'wheel_read') return jsonOutput_(readWheelBundle_());

    // Legacy dashboard behavior: a plain GET returns Activities as an array.
    return jsonOutput_(readActivities_());
  } catch (err) {
    return jsonOutput_({ error: String(err && err.message || err), codeVersion: CODE_VERSION });
  }
}

function doPost(e) {
  try {
    const body = e && e.postData && e.postData.contents ? JSON.parse(e.postData.contents) : {};
    const action = String(body.action || '').toLowerCase();
    if (action === 'wheel_write') return jsonOutput_(writeWheelBundle_(body));
    return jsonOutput_({ error: 'Unknown POST action: ' + action, codeVersion: CODE_VERSION });
  } catch (err) {
    return jsonOutput_({ error: String(err && err.message || err), codeVersion: CODE_VERSION });
  }
}

// ════════════════════════════════════════════════════
//   ACTIVITIES API
// ════════════════════════════════════════════════════

function setupActivitiesSheet_() {
  const sh = getOrCreateSheet_(SHEET_ACTIVITIES);
  if (sh.getLastRow() === 0) {
    sh.getRange(1, 1, 1, ACTIVITY_HEADERS.length).setValues([ACTIVITY_HEADERS]);
    formatTable_(sh, 1, ACTIVITY_HEADERS.length);
    sh.setFrozenRows(1);
  }
}

function readActivities_() {
  const sh = getSheet_(SHEET_ACTIVITIES);
  if (!sh || sh.getLastRow() < 2) return [];

  const values = sh.getDataRange().getValues();
  const headers = values.shift().map(h => normalizeHeader_(h));
  const rows = values.filter(row => row.some(cell => cell !== '' && cell !== null));
  const flows = [];

  rows.forEach((row, index) => {
    const obj = rowToObject_(headers, row);

    // Supports direct transaction format: date, asset, type, amount, currency, memo.
    if (obj.asset && obj.amount !== '') {
      const amount = parseNumber_(obj.amount);
      flows.push({
        id: parseNumber_(obj.id) || makeActivityId_(obj.date, index, obj.asset),
        date: formatDateValue_(obj.date),
        asset: String(obj.asset || '').toLowerCase(),
        type: String(obj.type || (amount < 0 ? 'withdraw' : 'deposit')).toLowerCase(),
        amount: Math.abs(amount),
        currency: obj.currency || 'THB',
        memo: obj.memo || obj.note || ''
      });
      return;
    }

    // Legacy wide format: date, stocks, bonds, crypto, gold, currency, note.
    ['stocks', 'bonds', 'crypto', 'gold'].forEach(asset => {
      const amount = parseNumber_(obj[asset]);
      if (!amount) return;
      flows.push({
        id: makeActivityId_(obj.date, index, asset),
        date: formatDateValue_(obj.date),
        asset: asset,
        type: amount >= 0 ? 'deposit' : 'withdraw',
        amount: Math.abs(amount),
        currency: obj.currency || 'THB',
        memo: obj.memo || obj.note || ''
      });
    });
  });

  return flows;
}

function appendActivityFromParams_(p) {
  setupActivitiesSheet_();
  const sh = getSheet_(SHEET_ACTIVITIES);
  sh.appendRow([
    p.date || '',
    p.stocks || '',
    p.bonds || '',
    p.crypto || '',
    p.gold || '',
    p.currency || 'THB',
    p.note || p.memo || ''
  ]);
  return { ok: true, codeVersion: CODE_VERSION, updatedAt: new Date().toISOString() };
}

// ════════════════════════════════════════════════════
//   WHEEL API
// ════════════════════════════════════════════════════

function readWheelBundle_() {
  return {
    codeVersion: CODE_VERSION,
    orders: readTableObjects_(SHEET_WHEEL_ORDERS),
    lots: readTableObjects_(SHEET_WHEEL_LOTS),
    history: readTableObjects_(SHEET_WHEEL_HISTORY)
  };
}

function writeWheelBundle_(body) {
  writeObjectRows_(SHEET_WHEEL_ORDERS, body.orders || [], WHEEL_ORDER_HEADERS);
  writeObjectRows_(SHEET_WHEEL_LOTS, body.lots || [], WHEEL_LOT_HEADERS);
  writeObjectRows_(SHEET_WHEEL_HISTORY, body.history || body.closed || [], WHEEL_ORDER_HEADERS);
  return { ok: true, codeVersion: CODE_VERSION, updatedAt: new Date().toISOString() };
}

// ════════════════════════════════════════════════════
//   SHEET HELPERS
// ════════════════════════════════════════════════════

function readTableObjects_(sheetName) {
  const sh = getSheet_(sheetName);
  if (!sh || sh.getLastRow() < 2) return [];
  const values = sh.getDataRange().getValues();
  const headers = values.shift().map(h => normalizeHeader_(h));
  return values
    .filter(row => row.some(cell => cell !== '' && cell !== null))
    .map(row => rowToObject_(headers, row));
}

function writeObjectRows_(sheetName, rows, headers) {
  const sh = getOrCreateSheet_(sheetName);
  sh.clearContents();
  sh.getRange(1, 1, 1, headers.length).setValues([headers]);

  const values = (rows || []).map(row => headers.map(header => {
    if (row[header] !== undefined && row[header] !== null) return row[header];
    if (header === 'updatedAt') return new Date().toISOString();
    return '';
  }));

  if (values.length) sh.getRange(2, 1, values.length, headers.length).setValues(values);
  formatTable_(sh, 1, headers.length);
  sh.setFrozenRows(1);
}

function rowToObject_(headers, row) {
  const obj = {};
  headers.forEach((header, i) => {
    obj[header] = normalizeCell_(row[i]);
  });
  return obj;
}

function normalizeHeader_(value) {
  return String(value || '').trim();
}

function normalizeCell_(value) {
  if (value instanceof Date) return formatDateValue_(value);
  if (value === null || value === undefined) return '';
  return value;
}

function parseNumber_(value) {
  if (value === null || value === undefined || value === '') return 0;
  if (typeof value === 'number') return value;
  const text = String(value).trim();
  const negative = text.startsWith('-') || (text.startsWith('(') && text.endsWith(')'));
  const cleaned = text.replace(/[^0-9.]/g, '');
  const num = parseFloat(cleaned) || 0;
  return negative ? -num : num;
}

function formatDateValue_(value) {
  if (!value) return '';
  if (value instanceof Date) {
    return Utilities.formatDate(value, Session.getScriptTimeZone(), 'yyyy-MM-dd');
  }
  const parsed = new Date(value);
  if (!isNaN(parsed.getTime())) {
    return Utilities.formatDate(parsed, Session.getScriptTimeZone(), 'yyyy-MM-dd');
  }
  return String(value);
}

function makeActivityId_(date, index, asset) {
  const base = Date.parse(formatDateValue_(date)) || Date.now();
  return base + (index * 10) + String(asset || '').length;
}

function getSheet_(name) {
  return SpreadsheetApp.getActiveSpreadsheet().getSheetByName(name);
}

function getOrCreateSheet_(name) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  return ss.getSheetByName(name) || ss.insertSheet(name);
}

function formatTable_(sh, headerRow, columnCount) {
  sh.getRange(headerRow, 1, 1, columnCount)
    .setFontWeight('bold')
    .setBackground('#1f2937')
    .setFontColor('#ffffff');
  sh.autoResizeColumns(1, columnCount);
}

function jsonOutput_(data) {
  return ContentService
    .createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}
