// ============================================================
//  UNIFIED APPS SCRIPT — Investment Dashboard + Wheel Strategy
//  GET  (no param)              → investRead
//  GET  ?action=activities_write → activitiesWrite via params
//  GET  ?action=settings_read   → settingsRead
//  GET  ?action=wheel_read      → wheelRead
//  GET  ?action=satellite_read  → satelliteRead
//  GET  ?action=satellite_benchmark → satelliteBenchmark
//  GET  ?action=satellite_refresh_prices → satelliteRefreshPrices
//  POST action=wheel_write      → wheelWrite
//  POST action=satellite_write  → satelliteWrite
// ============================================================

var SPREADSHEET_ID  = SpreadsheetApp.getActiveSpreadsheet().getId();
var TAB_ACTIVITIES  = "Activities";
var TAB_SETTING     = "Setting_data";
var TAB_WHL_ORDERS  = "Wheel_orders";
var TAB_WHL_LOTS    = "Wheel_lots";
var TAB_WHL_HISTORY = "Wheel_history";
var TAB_SAT_POSITIONS = "Satellite_positions";
var TAB_SAT_TRADES    = "Satellite_trades";

function doGet(e) {
  var p = (e && e.parameter) ? e.parameter : {};
  if (p.action === 'activities_write') return activitiesWriteFromParams(p);
  if (p.action === 'settings_read')    return settingsRead();
  if (p.action === 'wheel_read')       return wheelRead();
  if (p.action === 'satellite_read')   return satelliteRead();
  if (p.action === 'satellite_benchmark') return satelliteBenchmark(p);
  if (p.action === 'satellite_refresh_prices') return satelliteRefreshPrices(p);
  return investRead();
}

function doPost(e) {
  var data;
  try { data = JSON.parse(e.postData.contents); }
  catch (err) { return jsonOut({ error: 'Invalid JSON: ' + err.message }); }
  if (data.action === 'wheel_write') return wheelWrite(data);
  if (data.action === 'satellite_write') return satelliteWrite(data);
  return jsonOut({ error: 'Unknown action: ' + data.action });
}

// ============================================================
//  activitiesWriteFromParams
//  Reads flat GET params: date, stocks, bonds, crypto, gold, currency, note
// ============================================================
function activitiesWriteFromParams(p) {
  try {
    var ss    = SpreadsheetApp.openById(SPREADSHEET_ID);
    var sheet = ss.getSheetByName(TAB_ACTIVITIES);
    if (!sheet) return jsonOut({ error: "Sheet '" + TAB_ACTIVITIES + "' not found" });

    var dateStr  = p.date     || '';
    var stocks   = p.stocks   !== '' && p.stocks   !== undefined ? parseFloat(p.stocks)  : null;
    var bonds    = p.bonds    !== '' && p.bonds    !== undefined ? parseFloat(p.bonds)   : null;
    var crypto   = p.crypto   !== '' && p.crypto   !== undefined ? parseFloat(p.crypto)  : null;
    var gold     = p.gold     !== '' && p.gold     !== undefined ? parseFloat(p.gold)    : null;
    var currency = p.currency || 'THB';
    var note     = p.note     || '';

    if (!dateStr) return jsonOut({ error: 'Missing date' });

    // Duplicate check
    var existing     = sheet.getDataRange().getValues();
    var existingKeys = {};
    existing.forEach(function(row) {
      var rawDate = row[1]; // col B
      if (!rawDate) return;
      var d = rawDate instanceof Date ? rawDate : new Date(rawDate);
      if (isNaN(d.getTime())) return;
      var ds  = Utilities.formatDate(d, 'Asia/Bangkok', 'yyyy-MM-dd');
      var key = ds + '|' + [row[2], row[3], row[4], row[5]].join('|');
      existingKeys[key] = true;
    });

    var key = dateStr + '|' +
      [stocks !== null ? stocks : '', bonds !== null ? bonds : '',
       crypto !== null ? crypto : '', gold  !== null ? gold  : ''].join('|');

    if (existingKeys[key])
      return jsonOut({ ok: true, written: 0, note: 'duplicate' });

    // Find first empty row in col B, starting from row 4 (skip header rows 1-3)
    var startSearchRow = 4;
    var colB      = sheet.getRange(startSearchRow, 2, sheet.getMaxRows() - startSearchRow + 1, 1).getValues();
    var insertRow = sheet.getLastRow() + 1;
    for (var r = 0; r < colB.length; r++) {
      if (colB[r][0] === '' || colB[r][0] === null || colB[r][0] === undefined) {
        insertRow = startSearchRow + r; // 1-based row number
        break;
      }
    }

    // Quarter label
    var dt      = new Date(dateStr);
    var month   = dt.getMonth() + 1;
    var quarter = month <= 3 ? 'Q1' : month <= 6 ? 'Q2' : month <= 9 ? 'Q3' : 'Q4';

    var newRow = [
      quarter,
      dateStr,
      stocks !== null ? stocks : '',
      bonds  !== null ? bonds  : '',
      crypto !== null ? crypto : '',
      gold   !== null ? gold   : '',
      currency,
      note
    ];

    sheet.getRange(insertRow, 1, 1, 8).setValues([newRow]);
    return jsonOut({ ok: true, written: 1, startRow: insertRow });

  } catch (err) {
    return jsonOut({ error: err.message });
  }
}

// ============================================================
//  investRead
// ============================================================
function investRead() {
  try {
    var ss    = SpreadsheetApp.openById(SPREADSHEET_ID);
    var sheet = ss.getSheetByName(TAB_ACTIVITIES);
    if (!sheet) return jsonOut({ error: "Sheet '" + TAB_ACTIVITIES + "' not found" });

    var data = sheet.getDataRange().getValues();
    var COL  = { DATE:1, STOCKS:2, BONDS:3, CRYPTO:4, GOLD:5, CURRENCY:6, NOTE:7 };
    var transactions = [];
    // Use idCounter starting at 3000 to avoid collision with
    // hardcoded wheel history IDs (2001-2099) in the dashboard
    var idCounter = 3000;

    for (var i = 0; i < data.length; i++) {
      var row     = data[i];
      var rawDate = row[COL.DATE];

      // Skip empty or non-date cells (headers, blank rows)
      if (!rawDate) continue;
      var dateObj = rawDate instanceof Date ? rawDate : new Date(rawDate);
      if (isNaN(dateObj.getTime())) continue;

      // Format date using sheet timezone — keep original date, no clamping
      var dateStr  = Utilities.formatDate(dateObj, 'Asia/Bangkok', 'yyyy-MM-dd');

      var currency = String(row[COL.CURRENCY] || 'THB').trim().toUpperCase();
      if (currency !== 'USD') currency = 'THB';
      var note = row[COL.NOTE] ? String(row[COL.NOTE]).trim() : '';

      [
        { col: COL.STOCKS, asset: 'stocks' },
        { col: COL.BONDS,  asset: 'bonds'  },
        { col: COL.CRYPTO, asset: 'crypto' },
        { col: COL.GOLD,   asset: 'gold'   }
      ].forEach(function(m) {
        var val = row[m.col];

        // Skip truly empty cells
        if (val === '' || val === null || val === undefined) return;

        // Parse number — handle plain numbers, formatted strings,
        // and parenthesis-style negatives e.g. (42,881.61)
        var raw = String(val).trim();
        var isNegative = (raw.charAt(0) === '-') || (raw.charAt(0) === '(' && raw.charAt(raw.length - 1) === ')');
        var amount = parseFloat(raw.replace(/[^0-9.]/g, ''));
        if (isNaN(amount) || amount === 0) return;
        if (isNegative) amount = -amount;

        var txCurrency = (m.asset === 'bonds' || m.asset === 'gold') ? 'THB' : currency;
        transactions.push({
          id:       idCounter++,
          date:     dateStr,
          asset:    m.asset,
          type:     amount >= 0 ? 'deposit' : 'withdraw',
          amount:   Math.abs(amount),
          currency: txCurrency,
          memo:     note
        });
      });
    }
    return jsonOut(transactions);
  } catch (err) {
    return jsonOut({ error: err.message });
  }
}

// ============================================================
//  settingsRead
//  Reads Setting_data:
//  green table: Period, Stocks, Bonds, Crypto-THB, Crypto-O, Crypto-S, Gold, Exchange rate
//  blue table: Type, Class, Name, Ticker, Beg, Q1, Q2, Q3, Q4
// ============================================================
function settingsRead() {
  try {
    var ss    = SpreadsheetApp.openById(SPREADSHEET_ID);
    var sheet = ss.getSheetByName(TAB_SETTING);
    var out   = emptySettings();
    if (!sheet) return jsonOut(out);

    var values = sheet.getDataRange().getValues();
    readSettingBalances(values, out);
    readSettingTypedRows(values, out);
    return jsonOut(out);
  } catch (err) {
    return jsonOut({ error: err.message });
  }
}

function emptySettings() {
  return {
    begBalances: { stocks: 0, bonds: 0, crypto: 0, gold: 0 },
    endBalances: {
      stocks: { q1: null, q2: null, q3: null, q4: null },
      bonds:  { q1: null, q2: null, q3: null, q4: null },
      crypto: {
        q1: { thb: null, usdo: null, usds: null },
        q2: { thb: null, usdo: null, usds: null },
        q3: { thb: null, usdo: null, usds: null },
        q4: { thb: null, usdo: null, usds: null }
      },
      gold: { q1: null, q2: null, q3: null, q4: null }
    },
    rates: { beginning: null, q1: null, q2: null, q3: null, q4: null },
    benchmarks: {
      stocks: { name: '', q1: null, q2: null, q3: null, q4: null },
      bonds:  { name: '', q1: null, q2: null, q3: null, q4: null },
      crypto: { name: '', q1: null, q2: null, q3: null, q4: null },
      gold:   { name: '', q1: null, q2: null, q3: null, q4: null }
    },
    holdings: []
  };
}

function readSettingBalances(values, out) {
  var hRow = findSettingHeaderRow(values, 'period');
  if (hRow < 0) return;

  var h = values[hRow].map(normSettingText);
  var cPeriod = h.indexOf('period');
  var cStocks = findSettingCol(h, ['stocks']);
  var cBonds  = findSettingCol(h, ['bonds']);
  var cCThb   = findSettingCol(h, ['crypto thb', 'cryptothb', 'crypto']);
  var cCO     = findSettingCol(h, ['crypto o', 'cryptoo']);
  var cCS     = findSettingCol(h, ['crypto s', 'cryptos']);
  var cGold   = findSettingCol(h, ['gold']);
  var cRate   = findSettingCol(h, ['exchange rate', 'exchange rate usd thb', 'usd thb', 'rate']);

  for (var r = hRow + 1; r < values.length; r++) {
    var row = values[r];
    var period = normSettingPeriod(row[cPeriod]);
    if (!period) continue;

    if (period === 'beg') {
      out.begBalances.stocks = settingNumber(row[cStocks]) || 0;
      out.begBalances.bonds  = settingNumber(row[cBonds]) || 0;
      out.begBalances.crypto = settingNumber(row[cCThb]) || 0;
      out.begBalances.gold   = settingNumber(row[cGold]) || 0;
      out.rates.beginning    = settingNumber(row[cRate]);
      continue;
    }

    var q = period.toLowerCase();
    out.endBalances.stocks[q] = settingNumber(row[cStocks]);
    out.endBalances.bonds[q]  = settingNumber(row[cBonds]);
    out.endBalances.crypto[q] = {
      thb:  settingNumber(row[cCThb]),
      usdo: settingNumber(row[cCO]),
      usds: settingNumber(row[cCS])
    };
    out.endBalances.gold[q] = settingNumber(row[cGold]);
    out.rates[q] = settingNumber(row[cRate]);
  }
}

function readSettingTypedRows(values, out) {
  var hRow = findSettingHeaderRow(values, 'type');
  if (hRow < 0) return;

  var h = values[hRow].map(normSettingText);
  var cType   = h.indexOf('type');
  var cClass  = findSettingCol(h, ['class', 'asset class']);
  var cName   = findSettingCol(h, ['name']);
  var cTicker = findSettingCol(h, ['ticker']);
  var periodCols = {
    Beg: findSettingCol(h, ['beg', 'beginning']),
    Q1:  findSettingCol(h, ['q1']),
    Q2:  findSettingCol(h, ['q2']),
    Q3:  findSettingCol(h, ['q3']),
    Q4:  findSettingCol(h, ['q4'])
  };

  for (var r = hRow + 1; r < values.length; r++) {
    var row = values[r];
    var type = normSettingText(row[cType]);
    if (!type) continue;

    var asset = normSettingAsset(row[cClass]);
    if (!asset) continue;

    if (type === 'benchmark') {
      out.benchmarks[asset].name = String(row[cName] || '').trim();
      ['Q1', 'Q2', 'Q3', 'Q4'].forEach(function(period) {
        out.benchmarks[asset][period.toLowerCase()] = settingPercent(row[periodCols[period]]);
      });
    }

    if (type === 'holding') {
      ['Beg', 'Q1', 'Q2', 'Q3', 'Q4'].forEach(function(period) {
        var amount = settingNumber(row[periodCols[period]]);
        if (amount === null) return;
        out.holdings.push({
          asset: asset,
          period: period,
          name: String(row[cName] || '').trim(),
          ticker: String(row[cTicker] || '').trim(),
          amount: amount
        });
      });
    }
  }
}

function normSettingText(v) {
  return String(v || '').trim().toLowerCase()
    .replace(/[_/()]+/g, ' ')
    .replace(/[-]+/g, ' ')
    .replace(/\s+/g, ' ');
}

function normSettingPeriod(v) {
  var s = normSettingText(v);
  if (s === 'beg' || s === 'beginning') return 'beg';
  if (/^q[1-4]$/.test(s)) return s.toUpperCase();
  return '';
}

function normSettingAsset(v) {
  var s = normSettingText(v);
  if (s.indexOf('stock') === 0) return 'stocks';
  if (s.indexOf('bond') === 0) return 'bonds';
  if (s.indexOf('crypto') === 0) return 'crypto';
  if (s.indexOf('gold') === 0) return 'gold';
  return '';
}

function findSettingHeaderRow(values, name) {
  var target = normSettingText(name);
  for (var r = 0; r < values.length; r++) {
    if (values[r].map(normSettingText).indexOf(target) >= 0) return r;
  }
  return -1;
}

function findSettingCol(headers, names) {
  for (var i = 0; i < names.length; i++) {
    var idx = headers.indexOf(normSettingText(names[i]));
    if (idx >= 0) return idx;
  }
  return -1;
}

function settingNumber(v) {
  if (v === '' || v === null || v === undefined) return null;
  if (typeof v === 'number') return v;
  var s = String(v).trim();
  if (!s || /^n\/?a$/i.test(s)) return null;
  var neg = s.charAt(0) === '-' || (s.charAt(0) === '(' && s.charAt(s.length - 1) === ')');
  var n = parseFloat(s.replace(/[^0-9.]/g, ''));
  if (isNaN(n)) return null;
  return neg ? -n : n;
}

function settingPercent(v) {
  if (v === '' || v === null || v === undefined) return null;
  if (typeof v === 'number') return Math.abs(v) > 0 && Math.abs(v) <= 1 ? v * 100 : v;
  return settingNumber(v);
}

//  Satellite Trading Portfolio
//  The web app creates these tabs automatically on first push:
//  Satellite_positions and Satellite_trades
// ============================================================
var SAT_POSITION_HEADERS = [
  'id','ticker','name','side','status','openDate','closeDate','quantity',
  'avgCost','currentPrice','exitPrice','realizedPL','dividends','fees',
  'currency','fxRate','fxDate','fxSource','priceDate','priceSource','platform','takeProfit','stopLoss','tags',
  'thesis','review','updatedAt'
];

var SAT_TRADE_HEADERS = [
  'id','positionId','date','ticker','action','quantity','price','fees',
  'currency','fxRate','fxDate','fxSource','platform','note','tags','pairNote','reference','realizedPL','createdAt'
];

function satelliteRead() {
  try {
    var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    var positionSheet = ss.getSheetByName(TAB_SAT_POSITIONS);
    var tradeSheet = ss.getSheetByName(TAB_SAT_TRADES);
    return jsonOut({
      initialized: !!positionSheet && !!tradeSheet,
      positions: structuredSheetToJson(positionSheet, SAT_POSITION_HEADERS),
      trades: structuredSheetToJson(tradeSheet, SAT_TRADE_HEADERS)
    });
  } catch (err) {
    return jsonOut({ error: err.message });
  }
}

function satelliteWrite(data) {
  try {
    var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    writeStructuredSheet(ss, TAB_SAT_POSITIONS, SAT_POSITION_HEADERS, data.positions || []);
    writeStructuredSheet(ss, TAB_SAT_TRADES, SAT_TRADE_HEADERS, data.trades || []);
    return jsonOut({ ok: true, timestamp: new Date().toISOString() });
  } catch (err) {
    return jsonOut({ error: err.message });
  }
}

function satelliteYahooDailyCloses(symbol, start, end) {
  var from = Math.floor(new Date(start + 'T00:00:00Z').getTime() / 1000);
  var to = Math.floor(new Date(end + 'T23:59:59Z').getTime() / 1000) + 86400;
  var url = 'https://query1.finance.yahoo.com/v8/finance/chart/'
    + encodeURIComponent(symbol)
    + '?period1=' + from
    + '&period2=' + to
    + '&interval=1d&events=history';
  var response = UrlFetchApp.fetch(url, {
    muteHttpExceptions: true,
    headers: { 'User-Agent': 'Mozilla/5.0' }
  });
  if (response.getResponseCode() !== 200) throw new Error('Yahoo HTTP ' + response.getResponseCode());
  var json = JSON.parse(response.getContentText());
  var result = json && json.chart && json.chart.result && json.chart.result[0];
  var timestamps = result && result.timestamp ? result.timestamp : [];
  var closes = result && result.indicators && result.indicators.quote && result.indicators.quote[0]
    ? result.indicators.quote[0].close : [];
  var points = [];
  for (var i = 0; i < timestamps.length; i++) {
    var close = Number(closes[i]);
    if (!isFinite(close) || close <= 0) continue;
    var date = Utilities.formatDate(new Date(timestamps[i] * 1000), 'UTC', 'yyyy-MM-dd');
    if (date <= end) points.push({ date: date, close: close });
  }
  return points;
}

function satelliteRefreshPrices(p) {
  try {
    var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    var positionSheet = ss.getSheetByName(TAB_SAT_POSITIONS);
    if (!positionSheet) return jsonOut({ error: "Sheet '" + TAB_SAT_POSITIONS + "' not found" });
    var positions = structuredSheetToJson(positionSheet, SAT_POSITION_HEADERS);
    var openTickers = {};
    positions.forEach(function(position) {
      var ticker = String(position.ticker || '').trim().toUpperCase();
      var status = String(position.status || 'open').trim().toLowerCase();
      if (ticker && ticker !== 'CASH' && status !== 'closed') openTickers[ticker] = true;
    });
    var symbols = Object.keys(openTickers);
    if (!symbols.length) return jsonOut({ ok: true, updated: 0, quotes: {}, positions: positions });

    var end = /^\d{4}-\d{2}-\d{2}$/.test(String(p.end || '')) ? String(p.end) : Utilities.formatDate(new Date(), 'UTC', 'yyyy-MM-dd');
    var startDate = new Date(end + 'T00:00:00Z');
    var start = Utilities.formatDate(new Date(startDate.getTime() - 14 * 86400000), 'UTC', 'yyyy-MM-dd');
    var quotes = {};
    symbols.forEach(function(symbol) {
      try {
        var points = satelliteYahooDailyCloses(symbol, start, end);
        if (points.length) {
          var latest = points[points.length - 1];
          quotes[symbol] = { close: latest.close, date: latest.date, source: 'Yahoo Finance' };
        }
      } catch (err) {
        quotes[symbol] = { error: err.message };
      }
    });

    var updated = 0;
    positions = positions.map(function(position) {
      var ticker = String(position.ticker || '').trim().toUpperCase();
      var quote = quotes[ticker];
      if (!quote || !quote.close || !quote.date) return position;
      position.currentPrice = quote.close;
      position.priceDate = quote.date;
      position.priceSource = quote.source || 'Yahoo Finance';
      position.updatedAt = new Date().toISOString();
      updated++;
      return position;
    });
    if (updated) writeStructuredSheet(ss, TAB_SAT_POSITIONS, SAT_POSITION_HEADERS, positions);
    return jsonOut({ ok: true, updated: updated, quotes: quotes, positions: positions, fetchedAt: new Date().toISOString() });
  } catch (err) {
    return jsonOut({ error: 'Price refresh error: ' + err.message });
  }
}

function satelliteBenchmark(p) {
  try {
    var benchmark = p.benchmark === 'nasdaq' ? 'nasdaq' : 'sp500';
    var seriesId = benchmark === 'nasdaq' ? 'NASDAQCOM' : 'SP500';
    var name = benchmark === 'nasdaq' ? 'Nasdaq Composite' : 'S&P 500';
    var start = String(p.start || '').trim();
    var end = String(p.end || '').trim();
    if (!/^\d{4}-\d{2}-\d{2}$/.test(start) || !/^\d{4}-\d{2}-\d{2}$/.test(end)) {
      return jsonOut({ error: 'Invalid benchmark date range' });
    }

    var cache = CacheService.getScriptCache();
    var cacheKey = ['sat-benchmark-v2', benchmark, start, end].join(':');
    var cached = cache.get(cacheKey);
    if (cached) return jsonOut(JSON.parse(cached));

    var startDate = new Date(start + 'T00:00:00Z');
    var queryStart = Utilities.formatDate(new Date(startDate.getTime() - 7 * 86400000), 'UTC', 'yyyy-MM-dd');
    try {
      var yahooSymbol = benchmark === 'nasdaq' ? '^IXIC' : '^GSPC';
      var yahooPoints = satelliteYahooDailyCloses(yahooSymbol, queryStart, end);
      if (yahooPoints.length) {
        var yahooBasePoint = yahooPoints[0];
        for (var y = 0; y < yahooPoints.length; y++) {
          if (yahooPoints[y].date < start) yahooBasePoint = yahooPoints[y];
          else break;
        }
        var yahooDisplayPoints = yahooPoints.filter(function(point) { return point.date >= start; });
        if (yahooDisplayPoints.length) {
          var yahooBase = yahooBasePoint.close;
          var yahooOutput = {
            benchmark: benchmark,
            name: name,
            source: 'Yahoo Finance',
            start: start,
            end: end,
            updatedThrough: yahooPoints[yahooPoints.length - 1].date,
            points: yahooDisplayPoints.map(function(point) {
              return {
                date: point.date,
                close: point.close,
                returnPct: (point.close / yahooBase - 1) * 100
              };
            })
          };
          cache.put(cacheKey, JSON.stringify(yahooOutput), 21600);
          return jsonOut(yahooOutput);
        }
      }
    } catch (yahooErr) {
      // Fall back to FRED below. FRED can lag but is useful when Yahoo is unavailable.
    }

    var url = 'https://fred.stlouisfed.org/graph/fredgraph.csv?id='
      + encodeURIComponent(seriesId)
      + '&cosd=' + encodeURIComponent(queryStart)
      + '&coed=' + encodeURIComponent(end);
    var response = UrlFetchApp.fetch(url, {
      muteHttpExceptions: true
    });
    if (response.getResponseCode() !== 200) {
      return jsonOut({ error: 'Benchmark provider unavailable' });
    }

    var csvRows = Utilities.parseCsv(response.getContentText());
    if (!csvRows || csvRows.length < 2) return jsonOut({ error: 'Benchmark data unavailable' });
    var rawPoints = [];
    for (var i = 1; i < csvRows.length; i++) {
      var date = String(csvRows[i][0] || '').trim();
      var close = Number(csvRows[i][1]);
      if (!isFinite(close) || close <= 0) continue;
      if (date > end) continue;
      rawPoints.push({ date: date, close: close });
    }
    if (!rawPoints.length) return jsonOut({ error: 'No benchmark prices in selected range' });

    var basePoint = rawPoints[0];
    for (var j = 0; j < rawPoints.length; j++) {
      if (rawPoints[j].date < start) basePoint = rawPoints[j];
      else break;
    }
    var displayPoints = rawPoints.filter(function(point) { return point.date >= start; });
    if (!displayPoints.length) return jsonOut({ error: 'No benchmark prices in selected range' });
    var base = basePoint.close;
    var output = {
      benchmark: benchmark,
      name: name,
      source: 'FRED',
      start: start,
      end: end,
      updatedThrough: rawPoints[rawPoints.length - 1].date,
      points: displayPoints.map(function(point) {
        return {
          date: point.date,
          close: point.close,
          returnPct: (point.close / base - 1) * 100
        };
      })
    };
    cache.put(cacheKey, JSON.stringify(output), 21600);
    return jsonOut(output);
  } catch (err) {
    return jsonOut({ error: 'Benchmark error: ' + err.message });
  }
}

function structuredSheetToJson(sheet, expectedHeaders) {
  if (!sheet) return [];
  var values = sheet.getDataRange().getValues();
  if (values.length < 3) return [];
  var headers = values[2].map(function(h) { return String(h).trim(); });
  if (!headers.some(function(h) { return h; })) headers = expectedHeaders.slice();
  var rows = [];
  for (var r = 3; r < values.length; r++) {
    var row = values[r];
    if (!row.some(function(v) { return v !== '' && v !== null && v !== undefined; })) continue;
    var obj = {};
    for (var c = 0; c < headers.length; c++) {
      if (!headers[c]) continue;
      var value = row[c];
      if (value instanceof Date) value = Utilities.formatDate(value, Session.getScriptTimeZone(), 'yyyy-MM-dd');
      obj[headers[c]] = value == null ? '' : value;
    }
    rows.push(obj);
  }
  return rows;
}

function writeStructuredSheet(ss, tabName, headers, dataArray) {
  var sheet = ss.getSheetByName(tabName);
  if (!sheet) sheet = ss.insertSheet(tabName);
  sheet.getRange(1, 1).setValue(tabName.replace(/_/g, ' '));
  sheet.getRange(2, 1).setValue('Managed by Investment Dashboard. Row 3 contains field names.');
  sheet.getRange(3, 1, 1, headers.length).setValues([headers]);
  sheet.getRange(3, 1, 1, headers.length)
    .setFontWeight('bold')
    .setBackground('#D9EAD3')
    .setFontColor('#1F2937');

  var lastRow = sheet.getLastRow();
  var clearCols = Math.max(sheet.getLastColumn(), headers.length);
  if (lastRow >= 4) sheet.getRange(4, 1, lastRow - 3, clearCols).clearContent();
  if (!dataArray.length) return;

  var rows = dataArray.map(function(obj) {
    return headers.map(function(header) {
      var value = obj[header];
      return value === null || value === undefined ? '' : value;
    });
  });
  sheet.getRange(4, 1, rows.length, headers.length).setValues(rows);
  sheet.setFrozenRows(3);
}

function wheelRead() {
  try {
    var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    return jsonOut({
      orders:  sheetToJson(ss.getSheetByName(TAB_WHL_ORDERS)),
      lots:    sheetToJson(ss.getSheetByName(TAB_WHL_LOTS)),
      history: sheetToJson(ss.getSheetByName(TAB_WHL_HISTORY))
    });
  } catch (err) { return jsonOut({ error: err.message }); }
}

function wheelWrite(data) {
  try {
    var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    if (data.orders  !== undefined) writeSheet(ss, TAB_WHL_ORDERS,  data.orders);
    if (data.lots    !== undefined) writeSheet(ss, TAB_WHL_LOTS,    data.lots);
    if (data.history !== undefined) writeSheet(ss, TAB_WHL_HISTORY, data.history);
    return jsonOut({ ok: true, timestamp: new Date().toISOString() });
  } catch (err) { return jsonOut({ error: err.message }); }
}

function sheetToJson(sheet) {
  if (!sheet) return [];
  var values = sheet.getDataRange().getValues();
  if (values.length < 4) return [];
  var headers = values[2].map(function(h) { return String(h).trim(); });
  while (headers.length && headers[headers.length - 1] === '') headers.pop();
  var rows = [];
  for (var i = 3; i < values.length; i++) {
    var row = values[i];
    if (!row[0] || isNaN(Number(row[0])) || Number(row[0]) <= 0) continue;
    var obj = {};
    for (var j = 0; j < headers.length; j++) {
      if (!headers[j]) continue;
      var v = row[j];
      if (v instanceof Date) v = Utilities.formatDate(v, Session.getScriptTimeZone(), 'yyyy-MM-dd');
      obj[headers[j]] = v == null ? '' : v;
    }
    rows.push(obj);
  }
  return rows;
}

function writeSheet(ss, tabName, dataArray) {
  var sheet = ss.getSheetByName(tabName);
  if (!sheet || !dataArray || !dataArray.length) return;
  var headers = Object.keys(dataArray[0]);
  var lastRow  = sheet.getLastRow();
  if (lastRow >= 4) sheet.getRange(4, 1, lastRow - 3, sheet.getLastColumn()).clearContent();
  var rowData = dataArray.map(function(obj) {
    return headers.map(function(h) { var v = obj[h]; return v == null ? '' : v; });
  });
  sheet.getRange(4, 1, rowData.length, headers.length).setValues(rowData);
}

function jsonOut(obj) {
  var out = ContentService.createTextOutput(JSON.stringify(obj));
  out.setMimeType(ContentService.MimeType.JSON);
  return out;
}
